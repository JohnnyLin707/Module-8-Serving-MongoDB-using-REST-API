01

连接 MongoDB step-05-connect

起始

比上一步新增了什么

  • 用 dotenv 读取 .env 中的连接串，引入 Express 与 Mongoose。
  • start() 中先 await mongoose.connect()，成功后再 app.listen()；失败打印错误并退出进程。

为什么要这样写

  • 连接串属于配置信息，不能写死在源码里，否则易泄露且换环境要改代码。
  • 数据库连不上时启动 HTTP 服务没有意义，所以把「连接成功」作为监听的前置条件。

    console.log('Connected to MongoDB');

      console.log(`Book Store API listening on http://localhost:${port}`);
    });
  } catch (err) {
    console.error('Could not connect to MongoDB:', err.message);
    process.exit(1);

定义 Schema 与 Model step-06-schema-model

数据模型

比上一步新增了什么

  • 新增 bookSchema：title / author 必填且 trim，pages 必填且最小 1，fiction 默认 false，并开启 timestamps。
  • 用 mongoose.model('Book', bookSchema) 编译出操作集合的 Book 模型。

为什么要这样写

  • Schema 把数据规则集中声明一次，Mongoose 会在读写时自动校验、去空格、补默认值。
  • Model 提供 find/create/findByIdAndUpdate 等方法，是路由与数据库之间的接口。

种子数据脚本 step-07-seed

工具脚本

];

更新图书 PUT step-10-update

更新

比上一步新增了什么

  • 新增 PUT /api/books/:id，使用 findByIdAndUpdate。
  • 选项 returnDocument: 'after' 返回更新后的文档，runValidators: true 让更新也走 Schema 校验。

为什么要这样写

  • 一次数据库调用完成「更新 + 取回新值」，比先查再存更简洁。
  • Mongoose 默认更新不触发校验，显式开启才能防止写入不合法数据。

// routes/books.js: every /api/books route
const express = require('express');
const Book = require('../models/Book');

const router = express.Router();

// /api/books
router.route('/')
  .get(async (req, res) => {
    const filter = {};
    if (req.query.fiction) {
      filter.fiction = req.query.fiction === 'true';
    }
    const books = await Book.find(filter).sort({ title: 1 });
    res.json(books);
  })
  .post(async (req, res) => {
    const { title, author, pages, fiction } = req.body;
    const book = await Book.create({ title, author, pages, fiction });
    res.status(201).json(book);
  });

// /api/books/:id
router.route('/:id')
  .get(async (req, res) => {
    const book = await Book.findById(req.params.id);
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.json(book);
  })
  .put(async (req, res) => {
    const { title, author, pages, fiction } = req.body;
    const book = await Book.findByIdAndUpdate(
      req.params.id,
      { title, author, pages, fiction },
      { returnDocument: 'after', runValidators: true }
    );
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.json(book);
  })
  .delete(async (req, res) => {
    const book = await Book.findByIdAndDelete(req.params.id);
    if (!book) {
      return res.status(404).json({ message: 'Book not found.' });
    }
    res.status(204).end();
  });

module.exports = router;

// server.js: Book Store API (Express + MongoDB + Mongoose)
require('dotenv').config({ quiet: true });
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
const bookRoutes = require('./routes/books');

const app = express();
const port = process.env.PORT || 3000;

app.use(cors());            // let a React app on another port call this API
app.use(express.json());

app.get('/', (req, res) => {
  res.json({ message: 'Welcome to the Book Store API' });
});

app.use('/api/books', bookRoutes);   // every route in routes/books.js starts with /api/books

// ---------------------------------------------------------------- errors
// No route matched: 404
app.use((req, res, next) => {
  const err = new Error('Not Found');
  err.status = 404;
  next(err);
});

// The error handler (4 parameters): every error ends up here, as JSON
app.use((err, req, res, next) => {
  if (err.name === 'CastError') {
    return res.status(400).json({ error: { message: 'That is not a valid book id.' } });
  }
  if (err.name === 'ValidationError') {
    return res.status(400).json({ error: { message: err.message } });
  }
  res.status(err.status || 500).json({ error: { message: err.message } });
});

// ---------------------------------------------------------------- start
// Connect to MongoDB first; only then start listening for requests
async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI);
    console.log('Connected to MongoDB');
    app.listen(port, () => {
      console.log(`Book Store API listening on http://localhost:${port}`);
    });
  } catch (err) {
    console.error('Could not connect to MongoDB:', err.message);
    process.exit(1);
  }
}
